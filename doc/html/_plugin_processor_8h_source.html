125 // Programatically creates all parameters - called by AdaptiveMetronomeAudioProcessor-->apvts constructor
126 juce::AudioProcessorValueTreeState::ParameterLayout createParameterLayout()
127 {
128 float defaultAlpha = 0.1f;
129 float defaultBeta = 0.1f;
130 float defaultVolume = 1.0f;
131 float defaultTkNoise = 1.0f;
132 float defaultMNoise = 0.1f;
133 float defaultDelay = 0.0f;
134
135 juce::AudioProcessorValueTreeState::ParameterLayout params;
136
137 for (int i = 0; i < MAX_PLAYERS; i++)
138 {
139 // Volume
140 params.add(std::make_unique<AudioParameterFloatToUse>("player" + juce::String(i) + "-volume",
141 "Player " + juce::String(i) + " Volume",
142 0.0, 1.0, defaultVolume));
143
144 // Channel
145 params.add(std::make_unique<juce::AudioParameterInt>("player" + juce::String(i) + "-channel",
146 "Player " + juce::String(i) + " MIDI Channel",
147 1, 16, (i + 1)));
148
149 // Delay
150 params.add(std::make_unique<AudioParameterFloatToUse>("player" + juce::String(i) + "-delay",
151 "Player " + juce::String(i) + " Delay",
152 0.0, 200.0, defaultDelay));
153
154 // Motor Noise
155 params.add(std::make_unique<AudioParameterFloatToUse>("player" + juce::String(i) + "-mnoise-std",
156 "Player " + juce::String(i) + " Motor Noise Std",
157 0.0, 10.0, defaultMNoise));
158
159 // Timekeeper Noise
160 params.add(std::make_unique<AudioParameterFloatToUse>("player" + juce::String(i) + "-tknoise-std",
161 "Player " + juce::String(i) + " Time Keeper Noise Std",
162 0.0, 50.0, defaultTkNoise));
163
164 // Inner player loop for alphas
165 for (int j = 0; j < MAX_PLAYERS; j++)
166 {
167 // Alpha
168 params.add(std::make_unique<AudioParameterFloatToUse>("alpha-" + juce::String(i) + "-" + juce::String(j),
169 "Alpha " + juce::String(i) + "-" + juce::String(j),
170 0.0, 1.0, defaultAlpha));
171
172 // Beta
173 params.add(std::make_unique<AudioParameterFloatToUse>("beta-" + juce::String(i) + "-" + juce::String(j),
174 "Beta " + juce::String(i) + "-" + juce::String(j),
175 0.0, 1.0, defaultBeta));
176 }
177 }
178 return params;
179 }

181 //==============================================================================
182 // This is a helper class that simplifies access to parameter values.
183 // Using APVTS requires searching parameters by id.
184 // This allows access to parameters using bracket indexing.
185 // paramType can be AudioProcessorFloat or AudioProcessorInt
186 template <typename paramType>
187 class ParameterIndexGetter
188 {
189 private:
190 // Reference to the ValueTreeState to search for parameter.
191 juce::AudioProcessorValueTreeState &avpts;
192 // String that preceedes first player index in parameter name
193 juce::String before_I;
194 // String after first player index (and before second player index) in parameter name
195 juce::String after_I;
196
197 public:
198 ParameterIndexGetter(juce::AudioProcessorValueTreeState &apvts_ref,
199 juce::StringRef before_i,
200 juce::StringRef after_i) : avpts(apvts_ref),
201 before_I(juce::String(before_i)),
202 after_I(juce::String(after_i))
203 {
204 }
205
206 ~ParameterIndexGetter() {}
207
208 // Allows alphaParameter(i, j) type access to alpha and beta parameters
209 paramType *operator()(int i, int j)
210 {
211 juce::String matchString(before_I + juce::String(i) + after_I + juce::String(j));
212 auto audioParamPtr = dynamic_cast<paramType *>(avpts.getParameter(juce::StringRef(matchString)));
213 return audioParamPtr;
214 }
215
216 // Allows volumeParameter(i) type access to all other parameters
217 paramType *operator()(int i)
218 {
219 juce::String matchString(before_I + juce::String(i) + after_I);
220 auto audioParamPtr = dynamic_cast<paramType *>(avpts.getParameter(juce::StringRef(matchString)));
221 return audioParamPtr;
222 };
223 };
